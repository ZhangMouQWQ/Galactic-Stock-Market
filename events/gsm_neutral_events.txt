					check_variable = { which = gsm_t_clev value <= 10 }

				set_variable = { which = gsm_t_cm value = gsm_t_cap }
				divide_variable = { which = gsm_t_cm value = gsm_t_clev }

			set_variable = { which = gsm_t_cfee value = gsm_t_cn }
			multiply_variable = { which = gsm_t_cfee value = @gsm_cfd_fee }
			set_variable = { which = gsm_t_cpay value = gsm_t_cm }
			change_variable = { which = gsm_t_cpay value = gsm_t_cfee }
			if = {
				limit = {
					check_variable = { which = gsm_t_cm value >= @gsm_cfd_min_margin }
					check_variable = { which = gsm_t_ctr value >= gsm_t_cpay }
				}
				add_resource = { trade = -1 mult = gsm_t_cpay }
				gsm_cfd_mark_trader = yes
				event_target:gsm_t_tc = {
					set_variable = { which = gsm_t_iid value = prev.gsm_iid }
					set_variable = { which = gsm_t_cd value = prev.gsm_t_cdir }
					set_variable = { which = gsm_t_cm value = prev.gsm_t_cm }
					set_variable = { which = gsm_t_cl value = prev.gsm_t_clev }
					set_variable = { which = gsm_t_cn value = prev.gsm_t_cn }
					set_variable = { which = gsm_t_cfee value = prev.gsm_t_cfee }
					set_variable = { which = gsm_t_cai value = prev.gsm_t_cai }
					gsm_neutral_cfd_impact = { SIGN = 1 }
					set_variable = { which = gsm_t_ce value = gsm_price }
					set_variable = { which = gsm_t_cf value = 0 }
					set_variable = { which = gsm_t_co value = event_target:gsm_exchange.gsm_month }
