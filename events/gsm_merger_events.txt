country_event = {
	id = gsm_merger.4
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_variable = { which = gsm_t_mdone value = 0 }
		set_variable = { which = gsm_t_macc_ok value = 0 }
		gsm_company_controller = yes
		if = {
			limit = {
				check_variable = { which = gsm_t_has_ctl value = 1 }
				event_target:gsm_t_ctl = { is_same_value = event_target:gsm_t_macc }
				exists = event_target:gsm_off_by@this
			}
			event_target:gsm_off_by@this = { save_event_target_as = gsm_t_mby }
			event_target:gsm_t_acq = { gsm_company_controller = yes }
			if = {
				limit = {
					event_target:gsm_t_acq = { check_variable = { which = gsm_t_has_ctl value = 1 } }
					event_target:gsm_t_ctl = { is_same_value = event_target:gsm_t_mby }
				}
				remove_country_flag = gsm_off_pending
				gsm_merger_quote = yes
				if = {
					limit = { check_variable = { which = gsm_t_mq_ok value = 0 } }
					set_variable = { which = gsm_t_macc_ok value = 2 }
				}
				else_if = {
					limit = { check_variable = { which = gsm_t_mps value >= gsm_off_ps } }
					set_variable = { which = gsm_t_macc_ok value = 1 }
				}
			}
		}
		if = {
			limit = { check_variable = { which = gsm_t_macc_ok value = 1 } }
			gsm_merger_exec = yes
		}
		if = {
			limit = { check_variable = { which = gsm_t_mdone value = 1 } }
			set_variable = { which = gsm_off_last value = 0 }
		}
		else = {
			remove_country_flag = gsm_off_pending
			set_variable = { which = gsm_off_last value = 3 }
			if = {
				limit = { NOT = { check_variable = { which = gsm_t_macc_ok value = 0 } } }
				set_variable = { which = gsm_off_last value = 2 }
			}
		}
		if = {
			limit = { has_global_flag = gsm_debug_log }
			log = "GALMARKET_COMPANY: takeover offer for [This.GetRealName] accepted: result [This.gsm_t_mdone] check [This.gsm_t_macc_ok]"
		}
	}
}
