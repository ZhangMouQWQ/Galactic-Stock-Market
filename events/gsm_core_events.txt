event = {
	id = gsm_core.49
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_country = {
			limit = {
				is_country_type = default
				is_ai = yes
				OR = {
					has_policy_flag = gsm_tax_low
					has_policy_flag = gsm_tax_high
					has_policy_flag = gsm_fc_open
					has_policy_flag = gsm_fc_closed
					has_policy_flag = gsm_ir_laissez_faire
					has_policy_flag = gsm_ir_directed
					has_policy_flag = gsm_sub_public
					has_policy_flag = gsm_sub_all
					has_policy_flag = gsm_stx_std
					has_policy_flag = gsm_stx_high
					has_policy_flag = gsm_dout_on
					has_policy_flag = gsm_ofdi_limited
					has_policy_flag = gsm_ofdi_forbidden
				}
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_tax_low has_policy_flag = gsm_tax_high } }
				set_policy = { policy = gsm_policy_corporate_tax option = gsm_tax_standard cooldown = no }
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_fc_open has_policy_flag = gsm_fc_closed } }
				set_policy = { policy = gsm_policy_foreign_capital option = gsm_fc_restricted cooldown = no }
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_ir_laissez_faire has_policy_flag = gsm_ir_directed } }
				set_policy = { policy = gsm_policy_investment_regime option = gsm_ir_guided cooldown = no }
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_sub_public has_policy_flag = gsm_sub_all } }
				set_policy = { policy = gsm_policy_subsidies option = gsm_sub_none cooldown = no }
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_stx_std has_policy_flag = gsm_stx_high } }
				set_policy = { policy = gsm_policy_securities_tax option = gsm_stx_none cooldown = no }
			}
			if = {
				limit = { has_policy_flag = gsm_dout_on }
				set_policy = { policy = gsm_policy_defense_outsourcing option = gsm_dout_off cooldown = no }
			}
			if = {
				limit = { OR = { has_policy_flag = gsm_ofdi_limited has_policy_flag = gsm_ofdi_forbidden } }
				set_policy = { policy = gsm_policy_outward_fdi option = gsm_ofdi_regime cooldown = no }
			}
			if = {
				limit = { has_global_flag = gsm_debug_log }
				log = "GALMARKET: start policies reset to the defaults for [This.GetName]"
			}
		}
	}
}

		}
		gsm_gen_rb_colony_drop = yes
		gsm_gen_colony_reset = yes
		if = { limit = { is_variable_set = gsm_hkbl } clear_variable = gsm_hkbl }
		if = { limit = { is_variable_set = gsm_hkbw } clear_variable = gsm_hkbw }
		if = { limit = { is_variable_set = gsm_rbpm } clear_variable = gsm_rbpm }
		if = { limit = { is_variable_set = gsm_rbp_construction } clear_variable = gsm_rbp_construction }
		if = {
			limit = { OR = { has_planet_flag = gsm_has_assets is_variable_set = gsm_a_n } }
			save_event_target_as = gsm_t_cdp
			every_country = {
				limit = { gsm_is_live_company = yes event_target:gsm_t_cdp = { has_planet_flag = gsm_site_of@prev } }
				event_target:gsm_t_cdp = {
					set_variable = { which = gsm_t_s value = prev.gsm_slot }
					gsm_gen_asset_read = yes
				}
				set_variable = { which = gsm_t_wk value = event_target:gsm_t_cdp.gsm_t_rk }
				subtract_variable = { which = gsm_k value = gsm_t_wk }
				gsm_clamp_min = { VAR = gsm_k MIN = 0 }
				if = {
					limit = { has_global_flag = gsm_debug_log }
					log = "GALMARKET_ASSET: [This.GetRealName] lost its branch on [gsm_t_cdp.GetName] with the colony, capital [This.gsm_t_wk] written off"
				}
			}
			gsm_gen_planet_clear_all = yes
		}
	}
}
