# Financial Terminal window (market / portfolio / corps / ... pages).
# Deliberately NOT hidden from Message Settings: in a co-op empire every
# player shares one country, so one player paging through the terminal
# opens the next page for everyone. Players who do not want to be pulled
# along can open Message Settings and turn the popup off for this entry -
# the terminal then only pings the notification list, and the window opens
# on demand when they click it.

	default_popup = yes
