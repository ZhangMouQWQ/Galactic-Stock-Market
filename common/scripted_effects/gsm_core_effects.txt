gsm_clamp = {
	if = {
		limit = { check_variable = { which = $VAR$ value < $MIN$ } }
		set_variable = { which = $VAR$ value = $MIN$ }
	}
	if = {
		limit = { check_variable = { which = $VAR$ value > $MAX$ } }
		set_variable = { which = $VAR$ value = $MAX$ }
	}
}

	}
}

gsm_tick_sanctions = {
	set_variable = { which = gsm_t_sanc value = 0 }
	if = { limit = { is_active_resolution = resolution_sanctions_economic_1 } set_variable = { which = gsm_t_sanc value = 1 } }
	if = { limit = { is_active_resolution = resolution_sanctions_economic_2 } set_variable = { which = gsm_t_sanc value = 2 } }
	if = { limit = { is_active_resolution = resolution_sanctions_economic_3 } set_variable = { which = gsm_t_sanc value = 3 } }
	every_playable_country = {
		limit = { gsm_is_market_country = yes }
		gsm_default = { VAR = gsm_sanc VALUE = 0 }
		set_variable = { which = gsm_sanc_prev value = gsm_sanc }
		set_variable = { which = gsm_sanc value = 0 }
		if = {
			limit = {
				event_target:gsm_exchange = { check_variable = { which = gsm_t_sanc value > 0 } }
				is_galactic_community_member = yes
				OR = {
					is_in_breach_of_any = yes
					has_modifier = resolution_denounced
				}
			}
			set_variable = { which = gsm_sanc value = event_target:gsm_exchange.gsm_t_sanc }
		}
		if = {
			limit = { is_variable_set = gsm_sanc_force check_variable = { which = gsm_sanc value < gsm_sanc_force } }
			set_variable = { which = gsm_sanc value = gsm_sanc_force }
		}
		if = {
			limit = { check_variable = { which = gsm_sanc value > 0 } check_variable = { which = gsm_sanc_prev value = 0 } }
			gsm_news_push = { TYPE = 20 NOTIFY_ALL = yes }
		}
	}
}

gsm_news_push = {
	save_event_target_as = gsm_t_news_subject
	gsm_gen_mk_$TYPE$ = yes
	event_target:gsm_exchange = {
		set_variable = { which = gsm_news_5_type value = gsm_news_4_type }
		set_variable = { which = gsm_news_4_type value = gsm_news_3_type }
		set_variable = { which = gsm_news_3_type value = gsm_news_2_type }
		set_variable = { which = gsm_news_2_type value = gsm_news_1_type }
		set_variable = { which = gsm_news_1_type value = $TYPE$ }
		if = { limit = { is_variable_set = gsm_news_4_m } set_variable = { which = gsm_news_5_m value = gsm_news_4_m } }
		if = { limit = { is_variable_set = gsm_news_3_m } set_variable = { which = gsm_news_4_m value = gsm_news_3_m } }
		if = { limit = { is_variable_set = gsm_news_2_m } set_variable = { which = gsm_news_3_m value = gsm_news_2_m } }
		if = { limit = { is_variable_set = gsm_news_1_m } set_variable = { which = gsm_news_2_m value = gsm_news_1_m } }
		set_variable = { which = gsm_news_1_m value = gsm_month }
		if = { limit = { exists = event_target:gsm_news_4 } event_target:gsm_news_4 = { save_global_event_target_as = gsm_news_5 } }
		else_if = { limit = { exists = event_target:gsm_news_5 } clear_global_event_target = gsm_news_5 }
		if = { limit = { exists = event_target:gsm_news_3 } event_target:gsm_news_3 = { save_global_event_target_as = gsm_news_4 } }
		else_if = { limit = { exists = event_target:gsm_news_4 } clear_global_event_target = gsm_news_4 }
		if = { limit = { exists = event_target:gsm_news_2 } event_target:gsm_news_2 = { save_global_event_target_as = gsm_news_3 } }
		else_if = { limit = { exists = event_target:gsm_news_3 } clear_global_event_target = gsm_news_3 }
		if = { limit = { exists = event_target:gsm_news_1 } event_target:gsm_news_1 = { save_global_event_target_as = gsm_news_2 } }
		else_if = { limit = { exists = event_target:gsm_news_2 } clear_global_event_target = gsm_news_2 }
		event_target:gsm_t_news_subject = { save_global_event_target_as = gsm_news_1 }
		change_variable = { which = gsm_news_n value = 1 }
		gsm_clamp_max = { VAR = gsm_news_n MAX = 5 }
	}
	every_playable_country = {
		limit = { is_ai = no }
		save_event_target_as = gsm_t_reader
		if = {
			limit = {
				OR = {
					NOT = { event_target:gsm_t_news_subject = { gsm_is_security = yes } }
					event_target:gsm_t_news_subject = { gsm_visible_to_reader = yes }
				}
			}
			set_country_flag = gsm_news_unread
		}
	}
	[[NOTIFY_ALL]
	every_playable_country = {
		limit = { is_ai = no exists = capital_scope }
		save_event_target_as = gsm_t_reader
		if = {
			limit = {
				OR = {
					NOT = { event_target:gsm_t_news_subject = { gsm_is_security = yes } }
					event_target:gsm_t_news_subject = { gsm_visible_to_reader = yes }
				}
			}
			create_message = {
				type = GSM_MESSAGE_MARKET
				localization = GSM_NEWS_MSG_$TYPE$
				days = 30
				target = capital_scope
				variable = { type = name localization = COMPANY scope = event_target:gsm_t_news_subject }
			}
		}
	}
	]
	[[NOTIFY_STAKE]
	event_target:gsm_exchange = { save_event_target_as = gsm_t_news_grp }
	if = {
		limit = { event_target:gsm_t_news_subject = { has_country_flag = gsm_in_grp exists = event_target:gsm_grp@this } }
		event_target:gsm_t_news_subject = { event_target:gsm_grp@this = { save_event_target_as = gsm_t_news_grp } }
	}
	every_playable_country = {
		limit = {
			is_ai = no
			exists = capital_scope
			OR = {
				has_country_flag = gsm_holds@event_target:gsm_t_news_subject
				has_country_flag = gsm_holds@event_target:gsm_t_news_grp
			}
		}
		create_message = {
			type = GSM_MESSAGE_MARKET
			localization = GSM_NEWS_MSG_$TYPE$
			days = 30
			target = capital_scope
			variable = { type = name localization = COMPANY scope = event_target:gsm_t_news_subject }
		}
	}
	]
}
