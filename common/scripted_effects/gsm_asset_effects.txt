gsm_demand_factor = {
	set_variable = { which = $OUT$ value = 1 }
	set_variable = { which = gsm_t_dx value = $FREE$ }
	if = {
		limit = { check_variable = { which = gsm_t_dx value < 0 } }
		multiply_variable = { which = gsm_t_dx value = -1 }
		set_variable = { which = gsm_t_dy value = gsm_t_ps }
		multiply_variable = { which = gsm_t_dy value = @gsm_dmd_short_share }
		divide_variable = { which = gsm_t_dx value = gsm_t_dy }
		if = { limit = { check_variable = { which = gsm_t_dx value > 1 } } set_variable = { which = gsm_t_dx value = 1 } }
		multiply_variable = { which = gsm_t_dx value = @gsm_dmd_amp }
		change_variable = { which = $OUT$ value = gsm_t_dx }
	}
	else = {
		set_variable = { which = gsm_t_dy value = gsm_t_ps }
		multiply_variable = { which = gsm_t_dy value = @gsm_dmd_surplus_share }
		divide_variable = { which = gsm_t_dx value = gsm_t_dy }
		if = { limit = { check_variable = { which = gsm_t_dx value > 1 } } set_variable = { which = gsm_t_dx value = 1 } }
		multiply_variable = { which = gsm_t_dx value = @gsm_dmd_amp }
		subtract_variable = { which = $OUT$ value = gsm_t_dx }
	}
}
