gsm_sec_tax_calc = {
	if = {
		limit = {
			event_target:gsm_t_payer = { check_variable = { which = gsm_t_has_home value = 1 } }
			NOT = { is_same_value = event_target:gsm_t_home }
			event_target:gsm_t_home = { OR = { has_policy_flag = gsm_stx_std has_policy_flag = gsm_stx_high } }
		}
		if = {
			limit = { check_variable = { which = gsm_t_stax value < 0 } }
			set_variable = { which = gsm_t_stax value = 0 }
		}
		if = {
			limit = { event_target:gsm_t_home = { has_policy_flag = gsm_stx_high } }
			multiply_variable = { which = gsm_t_stax value = gsm_t_shigh }
		}
		else = {
			multiply_variable = { which = gsm_t_stax value = @gsm_stx_std }
		}
		if = {
			limit = { check_variable = { which = gsm_t_stax value > gsm_t_pay } }
			set_variable = { which = gsm_t_stax value = gsm_t_pay }
		}
		subtract_variable = { which = gsm_t_pay value = gsm_t_stax }
	}
	else = {
		set_variable = { which = gsm_t_stax value = 0 }
	}
}

gsm_company_pay_dividend = {
	save_event_target_as = gsm_t_payer
	set_variable = { which = gsm_t_pexact value = 0 }
	set_variable = { which = gsm_t_plim value = @gsm_mul_safe }
	divide_variable = { which = gsm_t_plim value = gsm_shares }
	if = {
		limit = { check_variable = { which = gsm_t_ptot value <= gsm_t_plim } }
		set_variable = { which = gsm_t_pexact value = 1 }
	}
	every_playable_country = {
		limit = { has_country_flag = gsm_holds@event_target:gsm_t_payer }
		event_target:gsm_t_payer = {
			set_variable = { which = gsm_t_iid value = prev.gsm_iid }
			gsm_gen_hold_get = yes
		}
		set_variable = { which = gsm_t_pay value = event_target:gsm_t_payer.gsm_t_hold }
		if = {
			limit = { event_target:gsm_t_payer = { check_variable = { which = gsm_t_pexact value = 1 } } }
			multiply_variable = { which = gsm_t_pay value = event_target:gsm_t_payer.gsm_t_ptot }
			divide_variable = { which = gsm_t_pay value = event_target:gsm_t_payer.gsm_shares }
		}
		else = {
			divide_variable = { which = gsm_t_pay value = event_target:gsm_t_payer.gsm_shares }
			multiply_variable = { which = gsm_t_pay value = event_target:gsm_t_payer.gsm_t_ptot }
		}
		set_variable = { which = gsm_t_stax value = gsm_t_pay }
		set_variable = { which = gsm_t_shigh value = @gsm_stx_div_high }
		gsm_sec_tax_calc = yes
		gsm_sec_tax_credit = yes
		if = {
			limit = { check_variable = { which = gsm_t_pay value > 0 } }
			add_resource = { trade = 1 mult = gsm_t_pay }
			gsm_default = { VAR = gsm_div_total VALUE = 0 }
			change_variable = { which = gsm_div_total value = gsm_t_pay }
			gsm_clamp_max = { VAR = gsm_div_total MAX = @gsm_total_cap }
		}
	}
	if = {
		limit = { has_country_flag = gsm_in_grp check_variable = { which = gsm_grp_h value > 0 } exists = event_target:gsm_grp@this }
		set_variable = { which = gsm_t_gpay value = gsm_grp_h }
		divide_variable = { which = gsm_t_gpay value = gsm_shares }
		multiply_variable = { which = gsm_t_gpay value = gsm_t_ptot }
		event_target:gsm_grp@this = {
			change_variable = { which = gsm_cash value = prev.gsm_t_gpay }
			gsm_default = { VAR = gsm_g_divq VALUE = 0 }
			change_variable = { which = gsm_g_divq value = prev.gsm_t_gpay }
		}
	}
	gsm_default = { VAR = gsm_t_pnofloat VALUE = 0 }
	set_variable = { which = gsm_t_fpay value = gsm_float }
	if = {
		limit = { check_variable = { which = gsm_t_pnofloat value = 1 } }
		set_variable = { which = gsm_t_fpay value = 0 }
	}
	if = {
		limit = { check_variable = { which = gsm_t_pexact value = 1 } }
		multiply_variable = { which = gsm_t_fpay value = gsm_t_ptot }
		divide_variable = { which = gsm_t_fpay value = gsm_shares }
	}
	else = {
		divide_variable = { which = gsm_t_fpay value = gsm_shares }
		multiply_variable = { which = gsm_t_fpay value = gsm_t_ptot }
	}
	if = {
		limit = {
			check_variable = { which = gsm_t_fpay value > 0 }
			check_variable = { which = gsm_t_has_home value = 1 }
		}
		event_target:gsm_t_home = {
			gsm_default = { VAR = gsm_pcap VALUE = 0 }
			change_variable = { which = gsm_pcap value = prev.gsm_t_fpay }
		}
	}
	else_if = {
		limit = {
			check_variable = { which = gsm_t_fpay value > 0 }
			check_variable = { which = gsm_t_has_home value = 0 }
		}
		change_variable = { which = gsm_cash value = gsm_t_fpay }
		if = {
			limit = { is_variable_set = gsm_dpa }
			change_variable = { which = gsm_dpa value = gsm_t_fpay }
		}
	}
}
