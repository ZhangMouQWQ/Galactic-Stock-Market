	if = {
		limit = { has_country_flag = gsm_ren_found }
		country_event = { id = gsm_ui.24 }
	}
	else_if = {
		limit = { has_country_flag = gsm_ren_gform }
		country_event = { id = gsm_ui.26 }
	}
	else_if = {
		limit = { has_country_flag = gsm_ren_grp }
		if = {
			limit = { exists = event_target:gsm_ren_co@root event_target:gsm_ren_co@root = { gsm_is_live_group = yes } }

			event_target:gsm_ren_co@root = { save_global_event_target_as = gsm_sel@root }
		}
		country_event = { id = gsm_ui.27 }
	}
	else_if = {
		limit = { has_country_flag = gsm_ren_active }
		country_event = { id = gsm_ui.15 }
	}
	else = {
		country_event = { id = gsm_ui.1 }
	}
}

gsm_ui_cfd_side_mode = {
	if = {
		limit = { check_variable = { which = gsm_ui_cfd_q value <= gsm_ui_cfd_$MAX$ } }
		if = {
			limit = { check_variable = { which = gsm_ui_cfd_has value = 0 } }
			set_variable = { which = gsm_ui_cfd_$M$ value = 1 }
			set_variable = { which = gsm_ui_cfd_$COST$ value = gsm_t_cost }
		}
		else_if = {
			limit = { check_variable = { which = gsm_ui_cfd_dir value = $SAME$ } }
			set_variable = { which = gsm_ui_cfd_$M$ value = 2 }
			set_variable = { which = gsm_ui_cfd_$COST$ value = gsm_t_cost }
		}
		else_if = {
			limit = { check_variable = { which = gsm_ui_cfd_q value < gsm_ui_cfd_pq } }
			set_variable = { which = gsm_ui_cfd_$M$ value = 3 }
			set_variable = { which = gsm_ui_cfd_$RET$ value = gsm_t_ra }
			multiply_variable = { which = gsm_ui_cfd_$RET$ value = gsm_ui_cfd_q }
			divide_variable = { which = gsm_ui_cfd_$RET$ value = gsm_ui_cfd_pq }
		}
		else_if = {
			limit = { check_variable = { which = gsm_ui_cfd_q value = gsm_ui_cfd_pq } }
			set_variable = { which = gsm_ui_cfd_$M$ value = 4 }
			set_variable = { which = gsm_ui_cfd_$RET$ value = gsm_t_ra }
		}
		else = {
			set_variable = { which = gsm_ui_cfd_$M$ value = 5 }
			set_variable = { which = gsm_ui_cfd_$RET$ value = gsm_t_ra }
			set_variable = { which = gsm_ui_cfd_$NEW$ value = gsm_ui_cfd_q }
			subtract_variable = { which = gsm_ui_cfd_$NEW$ value = gsm_ui_cfd_pq }
		}
	}
}

gsm_sit_start = {
	if = {
		limit = {
			is_ai = no
			gsm_is_market_country = yes
			NOT = { any_situation = { is_situation_type = gsm_market_sit } }
		}
		start_situation = { type = gsm_market_sit }
	}
}
gsm_sit_stop = {
	every_situation = {
		limit = { is_situation_type = gsm_market_sit }
		if = {
			limit = { has_global_flag = gsm_debug_log }
			log = "GALMARKET_UI: outliner situation removed"
		}
		destroy_situation = this
	}
}
gsm_sit_progress = {
	if = {
		limit = { exists = event_target:gsm_exchange }
		set_variable = { which = gsm_sit_p value = event_target:gsm_exchange.gsm_d_sent }
		set_situation_progress = gsm_sit_p
	}
}
gsm_sit_monthly = {
	every_playable_country = {
		limit = {
			OR = {
				has_edict = gsm_market_watch
				any_situation = { is_situation_type = gsm_market_sit }
			}
		}
		if = {
			limit = { has_edict = gsm_market_watch }
			gsm_sit_start = yes
			every_situation = {
				limit = { is_situation_type = gsm_market_sit }
				gsm_sit_progress = yes
			}
		}
		else = {
			gsm_sit_stop = yes
		}
	}
}
