gsm_rb_band = {
	set_variable = { which = gsm_t_bb value = -1 }
	set_variable = { which = gsm_t_bm value = 0 }
	set_variable = { which = gsm_t_bx value = gsm_t_bk }
	if = {
		limit = { check_variable = { which = gsm_t_bfl value = 1 } }
		set_variable = { which = gsm_t_bm value = 1 }
		if = { limit = { check_variable = { which = gsm_t_bx value < 0 } } set_variable = { which = gsm_t_bx value = 0 } }
	}
	else_if = {
		limit = { check_variable = { which = gsm_t_brv value <= -1 } }
		set_variable = { which = gsm_t_bm value = 0 }
	}
	else_if = {
		limit = { check_variable = { which = gsm_t_bk value >= @gsm_hurdle_roa } }
		set_variable = { which = gsm_t_bm value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { which = gsm_t_bsb value = 1 }
			check_variable = { which = gsm_t_bk value >= @gsm_subsidy_min_roa }
		}
		set_variable = { which = gsm_t_bm value = 2 }
		subtract_variable = { which = gsm_t_bx value = @gsm_subsidy_min_roa }
	}
	else_if = {
		limit = { check_variable = { which = gsm_t_bk value >= 0 } }
		set_variable = { which = gsm_t_bm value = 1 }
	}
	if = {
		limit = {
			check_variable = { which = gsm_t_bm value = 1 }
			check_variable = { which = gsm_t_bx value >= @gsm_hurdle_roa }
			event_target:gsm_exchange = { is_variable_set = gsm_rk_ok }
		}
		event_target:gsm_exchange = {
			set_variable = { which = gsm_t_hbx value = prev.gsm_t_bx }
			set_variable = { which = gsm_t_hbk value = prev.gsm_t_bki }
			country_event = { id = gsm_fn.62 }
		}
		set_variable = { which = gsm_t_bb value = event_target:gsm_exchange.gsm_t_hbb }
	}
	else_if = {
		limit = { check_variable = { which = gsm_t_bm value > 0 } }
		set_variable = { which = gsm_t_bb value = gsm_t_bx }
		divide_variable = { which = gsm_t_bb value = @gsm_band_w }
		gsm_floor = { VAR = gsm_t_bb }
		if = {
			limit = { check_variable = { which = gsm_t_bm value = 1 } check_variable = { which = gsm_t_bb value > @gsm_band_top } }
			set_variable = { which = gsm_t_bb value = @gsm_band_top }
		}
		if = {
			limit = { check_variable = { which = gsm_t_bm value = 2 } }
			if = {
				limit = { check_variable = { which = gsm_t_bb value >= @gsm_band_subn } }
				set_variable = { which = gsm_t_bb value = @gsm_band_subn }
				subtract_variable = { which = gsm_t_bb value = 1 }
			}
			change_variable = { which = gsm_t_bb value = @gsm_band_sub }
		}
	}
}
