		}

		change_variable = { which = gsm_realized_pl value = gsm_t_pay }
		subtract_variable = { which = gsm_realized_pl value = event_target:gsm_t_payer.gsm_t_cb }
		gsm_clamp = { VAR = gsm_realized_pl MIN = @gsm_total_floor MAX = @gsm_total_cap }
	}
	set_variable = { which = gsm_t_fpay value = gsm_float }
	if = {
		limit = { check_variable = { which = gsm_t_pexact value = 1 } }
		multiply_variable = { which = gsm_t_fpay value = gsm_t_ptot }
		divide_variable = { which = gsm_t_fpay value = gsm_shares }
	}
	else = {
		divide_variable = { which = gsm_t_fpay value = gsm_shares }
		multiply_variable = { which = gsm_t_fpay value = gsm_t_ptot }
	}
	if = {
		limit = { has_country_flag = gsm_in_grp check_variable = { which = gsm_grp_h value > 0 } exists = event_target:gsm_grp@this }
		set_variable = { which = gsm_t_gpay value = gsm_grp_h }
		divide_variable = { which = gsm_t_gpay value = gsm_shares }
		multiply_variable = { which = gsm_t_gpay value = gsm_t_ptot }
		event_target:gsm_grp@this = { change_variable = { which = gsm_cash value = prev.gsm_t_gpay } }
	}
	if = {
		limit = {
			check_variable = { which = gsm_t_fpay value > 0 }
			check_variable = { which = gsm_t_has_home value = 1 }
		}
		event_target:gsm_t_home = {
			gsm_default = { VAR = gsm_pcap VALUE = 0 }
			change_variable = { which = gsm_pcap value = prev.gsm_t_fpay }
		}
	}
	else_if = {
		limit = {
			check_variable = { which = gsm_t_fpay value > 0 }
			is_country_type = gsm_company
			is_variable_set = gsm_home_iid
			check_variable = { which = gsm_home_iid value >= 1 }
		}
		every_playable_country = {
			limit = {
				gsm_is_market_country = yes
				gsm_has_iid = yes
				check_variable = { which = gsm_iid value = event_target:gsm_t_payer.gsm_home_iid }
			}
			gsm_default = { VAR = gsm_pcap VALUE = 0 }
			change_variable = { which = gsm_pcap value = event_target:gsm_t_payer.gsm_t_fpay }
		}
	}
}
gsm_company_clear_holders = {
	save_event_target_as = gsm_t_payer
	every_playable_country = {
		limit = { has_country_flag = gsm_holds@event_target:gsm_t_payer }
		event_target:gsm_t_payer = {
			set_variable = { which = gsm_t_iid value = prev.gsm_iid }
			gsm_gen_hold_release = yes
		}
		remove_country_flag = gsm_holds@event_target:gsm_t_payer
	}
	set_variable = { which = gsm_foreign value = 0 }
}

	if = {
		limit = { check_variable = { which = gsm_defunct_m value >= @gsm_defunct_months } }
		gsm_company_destroy = yes
	}
}
