gsm_group_pick_name = {
	set_variable = { which = gsm_t_gn value = 0 }
	if = {
		limit = { exists = event_target:gsm_t_gfirst event_target:gsm_t_gfirst = { is_variable_set = gsm_name_p check_variable = { which = gsm_name_p value >= 1 } } }
		set_variable = { which = gsm_t_gn value = event_target:gsm_t_gfirst.gsm_name_p }
		change_variable = { which = gsm_t_gn value = -1 }
		modulo_variable = { which = gsm_t_gn value = @gsm_name_prefixes }
		change_variable = { which = gsm_t_gn value = 1 }
		if = {
			limit = { any_country = { NOT = { is_same_value = prev } gsm_is_live_group = yes check_variable = { which = gsm_gname_n value = prev.gsm_t_gn } } }
			set_variable = { which = gsm_t_gn value = 0 }
		}
	}
	set_variable = { which = gsm_t_gi value = 0 }
	while = {
		count = @gsm_group_names
		change_variable = { which = gsm_t_gi value = 1 }
		if = {
			limit = {
				check_variable = { which = gsm_t_gn value = 0 }
				NOT = { any_country = { NOT = { is_same_value = prev } gsm_is_live_group = yes check_variable = { which = gsm_gname_n value = prev.gsm_t_gi } } }
			}
			set_variable = { which = gsm_t_gn value = gsm_t_gi }
		}
	}
	gsm_clamp_min = { VAR = gsm_t_gn MIN = 1 }
	set_variable = { which = gsm_gname_n value = gsm_t_gn }
	gsm_gen_group_set_name = yes
}
gsm_group_apply_chosen_name = {
	set_variable = { which = gsm_t_gnc value = 0 }
	set_variable = { which = gsm_t_gnl value = 0 }
	if = {
		limit = {
			event_target:gsm_t_gfe = { is_ai = no has_country_flag = gsm_ren_gform }
			NOT = { gsm_ren_pl_valid = yes }
		}
		set_variable = { which = gsm_t_gnl value = 1 }
	}
	if = {
		limit = {
			event_target:gsm_t_gfe = { is_ai = no has_country_flag = gsm_ren_gform }
			gsm_ren_pl_valid = yes
		}
		set_variable = { which = gsm_t_gnc value = 1 }
	}
	else_if = {
		limit = { event_target:gsm_t_gfe = { is_ai = no is_variable_set = gsm_gf_np check_variable = { which = gsm_gf_np value >= 1 } } }
		set_variable = { which = gsm_t_gn value = event_target:gsm_t_gfe.gsm_gf_np }
		if = {
			limit = {
				check_variable = { which = gsm_t_gn value <= @gsm_group_names }
				NOT = { any_country = { NOT = { is_same_value = prev } gsm_is_live_group = yes check_variable = { which = gsm_gname_n value = prev.gsm_t_gn } } }
			}
			set_variable = { which = gsm_t_gnc value = 2 }
		}
		else = {
			set_variable = { which = gsm_t_gnl value = 1 }
		}
	}
	if = {
		limit = { check_variable = { which = gsm_t_gnc value = 1 } }
		set_name = event_target:gsm_ren_pl@root
		set_variable = { which = gsm_gname_n value = 0 }
	}
	else_if = {
		limit = { check_variable = { which = gsm_t_gnc value = 2 } }
		set_variable = { which = gsm_gname_n value = gsm_t_gn }
		gsm_gen_group_set_name = yes
	}
	else = {
		gsm_group_pick_name = yes
	}
	event_target:gsm_t_gfe = {
		if = {
			limit = { is_variable_set = gsm_gf_np }
			set_variable = { which = gsm_gf_np value = 0 }
		}
	}
}
