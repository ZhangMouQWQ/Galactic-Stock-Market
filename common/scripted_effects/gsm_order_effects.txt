			gsm_div_safe = { VAR = gsm_t_loq BY = gsm_t_locl }
			change_variable = { which = gsm_t_loq value = -1 }

				limit = { check_variable = { which = gsm_lo$I$_side value = 2 } }
				multiply_variable = { which = gsm_t_loq value = -1 }
			}
			gsm_clamp_max = { VAR = gsm_t_loq MAX = 1 }
			multiply_variable = { which = gsm_t_loq value = 2 }
			multiply_variable = { which = gsm_t_loq value = event_target:gsm_t_tc.gsm_shares }

			subtract_variable = { which = gsm_t_lorem value = gsm_lo$I$_done }
			gsm_clamp_max = { VAR = gsm_t_loq MAX = gsm_t_lorem }
			if = {
				limit = { check_variable = { which = gsm_lo$I$_side value = 1 } }
				export_resource_stockpile_to_variable = { resource = trade variable = gsm_t_lob }
				gsm_trade_fee_rate = yes
				set_variable = { which = gsm_t_lox value = gsm_t_feer }
				change_variable = { which = gsm_t_lox value = 1 }
				multiply_variable = { which = gsm_t_lox value = gsm_lo$I$_px }
				gsm_div_safe = { VAR = gsm_t_lob BY = gsm_t_lox }
				gsm_clamp_max = { VAR = gsm_t_loq MAX = gsm_t_lob }
			}
			gsm_floor = { VAR = gsm_t_loq }
			if = {
				limit = { check_variable = { which = gsm_t_loq value >= 1 } }
				set_variable = { which = gsm_t_oq value = gsm_t_loq }
				set_variable = { which = gsm_t_lotf value = 0 }
				set_variable = { which = gsm_t_trade_ok value = 0 }
				if = {
					limit = { check_variable = { which = gsm_lo$I$_side value = 1 } }
					gsm_trade_buy = yes
				}
				else = {
					gsm_trade_sell = yes
				}
				set_variable = { which = gsm_t_oq value = 0 }
				if = {
					limit = { check_variable = { which = gsm_t_trade_ok value = 1 } }
					change_variable = { which = gsm_lo$I$_done value = gsm_t_trade_q }
					change_variable = { which = gsm_t_lofill value = 1 }
					event_target:gsm_t_tc = { save_event_target_as = gsm_t_lolast }
				}
			}
			if = {
				limit = { check_variable = { which = gsm_lo$I$_done value >= gsm_lo$I$_q } }
				set_variable = { which = gsm_lo$I$_st value = 3 }
			}
			else_if = {
				limit = { check_variable = { which = gsm_lo$I$_done value > 0 } }
				set_variable = { which = gsm_lo$I$_st value = 2 }
			}
		}
	}
	if = {
		limit = {
			check_variable = { which = gsm_lo$I$_st value >= 1 }
			check_variable = { which = gsm_lo$I$_st value <= 2 }
		}
